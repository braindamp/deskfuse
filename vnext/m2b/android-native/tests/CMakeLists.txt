# Host-side deterministic clock qualification, no Android/AAudio runtime.
cmake_minimum_required(VERSION 3.24)
project(DeskFuseClockQualification LANGUAGES CXX)
enable_testing()
add_executable(cp001_controller controller_qualification.cpp)
target_compile_features(cp001_controller PRIVATE cxx_std_20)
target_include_directories(cp001_controller PRIVATE ../src)
if(MSVC)
  target_compile_options(cp001_controller PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(cp001_controller PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME cp001_controller COMMAND cp001_controller)

add_executable(mp001_linear_asrc_comparator asrc_comparator.cpp)
target_compile_features(mp001_linear_asrc_comparator PRIVATE cxx_std_20)
target_include_directories(mp001_linear_asrc_comparator PRIVATE ../src)
if(MSVC)
  target_compile_options(mp001_linear_asrc_comparator PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(mp001_linear_asrc_comparator PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME mp001_linear_asrc_comparator COMMAND mp001_linear_asrc_comparator)

find_package(Threads REQUIRED)
add_executable(mp001_actual_render_core live_render_core.cpp)
target_compile_features(mp001_actual_render_core PRIVATE cxx_std_20)
target_include_directories(mp001_actual_render_core PRIVATE ../src)
target_compile_definitions(mp001_actual_render_core PRIVATE DESKFUSE_TEST_PROBE=1)
target_link_libraries(mp001_actual_render_core PRIVATE Threads::Threads)
if(MSVC)
  target_compile_options(mp001_actual_render_core PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(mp001_actual_render_core PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME mp001_actual_render_core COMMAND mp001_actual_render_core)
