cmake_minimum_required(VERSION 3.20)
project(DeskFuseM2aWire LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  add_compile_options(/W4 /permissive- /EHsc)
else()
  add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
add_executable(df_wire_tests tests/test_wire.cpp)
target_include_directories(df_wire_tests PRIVATE include ../wp001/include)
enable_testing()
add_test(NAME m2a_wire_contract COMMAND df_wire_tests)
