cmake_minimum_required(VERSION 3.24)
project(DeskFuseM1Wasapi LANGUAGES CXX)

if(NOT WIN32)
  message(FATAL_ERROR "M1 WASAPI capture is a Windows-only platform adapter")
endif()

add_executable(deskfuse-wasapi-capture src/wasapi_capture.cpp)
target_compile_features(deskfuse-wasapi-capture PRIVATE cxx_std_20)
target_compile_definitions(deskfuse-wasapi-capture PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(deskfuse-wasapi-capture PRIVATE ole32 uuid propsys avrt)
add_library(deskfuse-source-control SHARED src/source_control.cpp)
target_compile_features(deskfuse-source-control PRIVATE cxx_std_20)
target_compile_definitions(deskfuse-source-control PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(deskfuse-source-control PRIVATE ole32 uuid propsys avrt)
if(MSVC)
  target_compile_options(deskfuse-wasapi-capture PRIVATE /W4 /permissive- /EHsc)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(deskfuse-pcm-credits-test tests/pcm_credits_test.cpp)
  target_compile_features(deskfuse-pcm-credits-test PRIVATE cxx_std_20)
  add_test(NAME pcm-credit-ownership COMMAND deskfuse-pcm-credits-test)
  add_executable(deskfuse-frame-clock-test tests/frame_clock_test.cpp)
  target_compile_features(deskfuse-frame-clock-test PRIVATE cxx_std_20)
  add_test(NAME wavert-frame-clock COMMAND deskfuse-frame-clock-test)
  add_executable(deskfuse-wasapi-queue-test tests/spsc_capture_queue_test.cpp)
  target_compile_features(deskfuse-wasapi-queue-test PRIVATE cxx_std_20)
  target_compile_definitions(deskfuse-wasapi-queue-test PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN DESKFUSE_QUEUE_TEST_PROBE)
  if(MSVC)
    target_compile_options(deskfuse-wasapi-queue-test PRIVATE /W4 /permissive- /EHsc)
  endif()
  add_test(NAME wasapi-spsc-capture-queue COMMAND deskfuse-wasapi-queue-test)
  set_tests_properties(wasapi-spsc-capture-queue PROPERTIES TIMEOUT 20)
  add_executable(deskfuse-wasapi-writer-test tests/capture_writer_lifecycle_test.cpp)
  target_compile_features(deskfuse-wasapi-writer-test PRIVATE cxx_std_20)
  target_compile_definitions(deskfuse-wasapi-writer-test PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(deskfuse-wasapi-writer-test PRIVATE ole32 uuid propsys avrt)
  if(MSVC)
    target_compile_options(deskfuse-wasapi-writer-test PRIVATE /W4 /permissive- /EHsc)
  endif()
  add_test(NAME wasapi-writer-lifecycle COMMAND deskfuse-wasapi-writer-test)
  set_tests_properties(wasapi-writer-lifecycle PROPERTIES TIMEOUT 10)
endif()
