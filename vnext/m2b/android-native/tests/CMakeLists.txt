# Host-side deterministic clock qualification, no Android/AAudio runtime.
cmake_minimum_required(VERSION 3.24)
project(DeskFuseClockQualification LANGUAGES CXX)
enable_testing()
add_executable(cp001_controller controller_qualification.cpp)
target_compile_features(cp001_controller PRIVATE cxx_std_20)
target_include_directories(cp001_controller PRIVATE ../src)
if(MSVC)
  target_compile_options(cp001_controller PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(cp001_controller PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME cp001_controller COMMAND cp001_controller)

add_executable(mp001_linear_asrc_comparator asrc_comparator.cpp)

option(DESKFUSE_QUALIFY_LIBSAMPLERATE "Run first persistent SRC candidate host tests" OFF)
if(DESKFUSE_QUALIFY_LIBSAMPLERATE)
  find_path(LIBSAMPLERATE_HEADERS samplerate.h REQUIRED)
  find_library(LIBSAMPLERATE_LIB samplerate REQUIRED)
  add_executable(mp001_libsamplerate_streaming libsamplerate_streaming.cpp)
  target_compile_features(mp001_libsamplerate_streaming PRIVATE cxx_std_20)
  target_include_directories(mp001_libsamplerate_streaming PRIVATE ../src "${LIBSAMPLERATE_HEADERS}")
  target_link_libraries(mp001_libsamplerate_streaming PRIVATE "${LIBSAMPLERATE_LIB}")
  if(MSVC)
    target_compile_options(mp001_libsamplerate_streaming PRIVATE /W4 /WX /permissive-)
  else()
    target_compile_options(mp001_libsamplerate_streaming PRIVATE -Wall -Wextra -Werror)
  endif()
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_library(df_allocation_probe SHARED allocation_probe_linux.cpp)
    target_compile_features(df_allocation_probe PRIVATE cxx_std_20)
    # Strong interposition through LD_PRELOAD catches dynamic libsamplerate
    # libc allocations, not merely allocations in our caller's compilation unit.
    target_link_libraries(mp001_libsamplerate_streaming PRIVATE df_allocation_probe)
    target_compile_definitions(mp001_libsamplerate_streaming PRIVATE DESKFUSE_TEST_ALLOC_PROBE=1)
  endif()
  add_test(NAME mp001_libsamplerate_streaming COMMAND mp001_libsamplerate_streaming)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set_tests_properties(mp001_libsamplerate_streaming PROPERTIES
      ENVIRONMENT "LD_PRELOAD=$<TARGET_FILE:df_allocation_probe>")
  endif()
endif()
